--•	Cadastro de Colaboradores (Código, Nome, E-mail, Cargo e Situação).

CREATE TABLE colaboradores(

	  id_colaborador INT PRIMARY KEY IDENTITY(1,1),
	  codigo_colaborador INT NOT NULL UNIQUE,
	  nome_colaborador varchar(80)not null,
	  email_colaborador varchar(80) not null, 
	  cargo_colaborador varchar(80)not null ,
	  situacao_colaborador varchar(80) not null,
	  senha_colaborador varchar(80)not null
);

--•	Cadastro de Projetos (Código, Nome, Descrição, Datas e Status).

CREATE TABLE Projetos(

	  id_projeto INT PRIMARY KEY IDENTITY(1,1),
	  codigo_projeto INT NOT NULL UNIQUE,
	  nome_projeto varchar(80)not null,
	  descricao_projeto varchar(80) not null, 
	  data_criacao varchar(80)not null,
	  data_conclusao varchar(80),
	  status_projeto varchar(80) not null,
);

--•	Cadastro de Tarefas (Projeto, Responsável, Descrição, Prioridade, Datas e Situação).

CREATE TABLE Tarefas (
    id_tarefa INT IDENTITY(1,1) PRIMARY KEY,
    ProjetoId INT NOT NULL,           -- Seleção de "Lista de Projetos"
    ColaboradorId INT NOT NULL,       -- Seleção de "Lista de Colaboradores"
    Descricao VARCHAR(255) NOT NULL,  -- Campo "Descrição"
    Prioridade VARCHAR(20) NOT NULL,  -- Seleção de "Prioridade" (ALTA, MEDIA, BAIXA)
    Situacao VARCHAR(30) NOT NULL DEFAULT 'PENDENTE', -- Situação/Status da tarefa
    DataInicial DATE NOT NULL,        -- Campo "Data inicial"
    DataFinal DATE NULL               -- Campo "Data final"
);
